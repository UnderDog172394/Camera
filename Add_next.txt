static int g_fast_threshold = 40;

typedef struct { uint16_t x, y, score; } Keypoint;

typedef int (*cmp_fn)(const void*, const void*);
static int cmp_kp_desc(const void* a, const void* b){
    const Keypoint *ka = (const Keypoint*)a, *kb = (const Keypoint*)b;
    return (int)kb->score - (int)ka->score; // higher first
}

static int grid_select(const Keypoint *in, int n, int w, int h,
                       int gx, int gy, int cell_max, Keypoint *out)
{
    const int cw = (w + gx - 1) / gx;
    const int ch = (h + gy - 1) / gy;
    static uint8_t used[GRID_Y][GRID_X];
    memset(used, 0, sizeof(used));
    int kept = 0;
    for (int i = 0; i < n; ++i) {
        int cx = in[i].x / cw;
        int cy = in[i].y / ch;
        if ((unsigned)cx >= (unsigned)gx || (unsigned)cy >= (unsigned)gy) continue;
        if (used[cy][cx] < cell_max) {
            out[kept++] = in[i];
            used[cy][cx]++;
        }
    }
    return kept;
}

// ----------------------------- FAST CORE ------------------------------------
static void fast_detect_and_nms(const uint8_t *img, int w, int h, int *out_count)
{
    if (!score_grid) { *out_count = 0; return; }
    memset(score_grid, 0, w*h*sizeof(uint16_t));

    const int base_t = g_fast_threshold;
    const int8_t ox[16] = { 0,1,2,3,3,3,2,1,0,-1,-2,-3,-3,-3,-2,-1 };
    const int8_t oy[16] = { 3,3,2,1,0,-1,-2,-3,-3,-3,-2,-1,0,1,2,3 };
    const int qi[4]     = { 1,5,9,13 };

    for (int y=3; y<h-3; ++y) {
        const int yw = y*w;
        int x0 = 3;
        if (FAST_STRIDE == 2) x0 += (y & 1); // checkerboard

        for (int x = x0; x < w-3; x += FAST_STRIDE) {
            const uint8_t p = img[yw + x];
            int t = base_t;
            const int hi = (p + t > 255) ? 255 : (p + t);
            const int lo = (p - t <   0) ?   0 : (p - t);

            int bright=0, dark=0;
            for (int k=0;k<4;++k) {
                const int idx = qi[k];
                const uint8_t v = img[(y+oy[idx])*w + (x+ox[idx])];
                bright += (v>hi);
                dark   += (v<lo);
            }
            if (bright < 3 && dark < 3) continue;

            bool corner=false;
            for (int sidx=0; sidx<16 && !corner; ++sidx) {
                if (img[(y+oy[sidx])*w + (x+ox[sidx])] > hi) {
                    int run=1; for (int k=1;k<FAST_MIN_CONTIG;++k){
                        const int idx=(sidx+k)&0xF;
                        if (img[(y+oy[idx])*w + (x+ox[idx])] > hi) ++run; else break;
                    }
                    if (run>=FAST_MIN_CONTIG) corner=true;
                }
                if (!corner && img[(y+oy[sidx])*w + (x+ox[sidx])] < lo) {
                    int run=1; for (int k=1;k<FAST_MIN_CONTIG;++k){
                        const int idx=(sidx+k)&0xF;
                        if (img[(y+oy[idx])*w + (x+ox[idx])] < lo) ++run; else break;
                    }
                    if (run>=FAST_MIN_CONTIG) corner=true;
                }
            }
            if (!corner) continue;

            int score=0;
            for (int i=0;i<16;++i){
                const int v = img[(y+oy[i])*w + (x+ox[i])];
                score += (v>p)? (v-p):(p-v);
            }
            score_grid[yw + x] = (uint16_t)((score>0xFFFF)?0xFFFF:score);
        }
    }

    int keep=0;
    for (int y=3;y<h-3;++y){
        const int yw=y*w;
        for (int x=3;x<w-3;++x){
            const uint16_t s = score_grid[yw+x]; if(!s) continue;
            if (s > score_grid[yw-w + x-1] && s > score_grid[yw-w + x] && s > score_grid[yw-w + x+1] &&
                s > score_grid[yw   + x-1] &&                             s > score_grid[yw   + x+1] &&
                s > score_grid[yw+w + x-1] && s > score_grid[yw+w + x] && s > score_grid[yw+w + x+1]) {
                if (keep<MAX_KEYPOINTS)
                    keypoints[keep++] = (Keypoint){(uint16_t)x,(uint16_t)y,s};
            }
        }
    }

    qsort(keypoints, keep, sizeof(Keypoint), (cmp_fn)cmp_kp_desc);

#if TILED_NMS
    static Keypoint tmp[MAX_KEYPOINTS];
    int kept = grid_select(keypoints, keep, g_w, g_h, GRID_X, GRID_Y, CELL_MAX, tmp);
    if (kept > 0) memcpy(keypoints, tmp, kept * sizeof(Keypoint));
    *out_count = kept;
#else
    *out_count = keep;
#endif
}

static esp_err_t camera_init_gray(void)
{
    camera_config_t cfg = {
        .pin_pwdn     = CAM_PIN_PWDN,
        .pin_reset    = CAM_PIN_RESET,
        .pin_xclk     = CAM_PIN_XCLK,
        .pin_sccb_sda = CAM_PIN_SIOD,
        .pin_sccb_scl = CAM_PIN_SIOC,
        .pin_d7 = CAM_PIN_D7, .pin_d6 = CAM_PIN_D6, .pin_d5 = CAM_PIN_D5, .pin_d4 = CAM_PIN_D4,
        .pin_d3 = CAM_PIN_D3, .pin_d2 = CAM_PIN_D2, .pin_d1 = CAM_PIN_D1, .pin_d0 = CAM_PIN_D0,
        .pin_vsync = CAM_PIN_VSYNC, .pin_href = CAM_PIN_HREF, .pin_pclk = CAM_PIN_PCLK,
        .xclk_freq_hz = 20000000,
        .ledc_timer   = LEDC_TIMER_0,
        .ledc_channel = LEDC_CHANNEL_0,
        .pixel_format = PIXFORMAT_GRAYSCALE,
        .frame_size   = FRAMESIZE_QVGA,
        .jpeg_quality = 12,
        .fb_count     = 2,
        .fb_location  = CAMERA_FB_IN_PSRAM,
        .grab_mode    = CAMERA_GRAB_WHEN_EMPTY
    };
    ESP_RETURN_ON_ERROR(esp_camera_init(&cfg), TAG, "camera init failed");

    camera_fb_t *fb = esp_camera_fb_get();
    if (!fb) return ESP_FAIL;
    g_w = fb->width; g_h = fb->height;
    esp_camera_fb_return(fb);

    score_grid = heap_caps_calloc(g_w*g_h, sizeof(uint16_t), MALLOC_CAP_SPIRAM);
    if (!score_grid) return ESP_ERR_NO_MEM;

    g_prev = heap_caps_malloc(g_w*g_h, MALLOC_CAP_SPIRAM);
    if (!g_prev) return ESP_ERR_NO_MEM;
    memset(g_prev, 0, g_w*g_h);

    ESP_LOGI(TAG, "Cam: %dx%d GRAYSCALE, stride=%d, gridNMS=%s (%dx%d,K=%d)",
         g_w, g_h, FAST_STRIDE, TILED_NMS? "ON":"OFF", GRID_X, GRID_Y, CELL_MAX);
    return ESP_OK;
}

    if (s->set_gain_ctrl)     s->set_gain_ctrl(s,     agc ? 1 : 0);
}

}

// ----------------------- SWEEP (same as before, terser) ---------------------
typedef struct { float avg; float std; int zero; int aec; int agc; } sweep_stat_t;

static void run_frames_eval(int nframes, sweep_stat_t *st)
{
    for (int i=0;i<2;i++){ camera_fb_t* f=esp_camera_fb_get(); if(f) esp_camera_fb_return(f); }
    double sum=0, sum2=0; int zero=0;
    for (int i=0;i<nframes;i++){
        camera_fb_t *fb = esp_camera_fb_get();
        if (!fb) { vTaskDelay(pdMS_TO_TICKS(2)); i--; continue; }
        int n_kp=0;
        fast_detect_and_nms(fb->buf, g_w, g_h, &n_kp);
        sum += n_kp; sum2 += (double)n_kp*n_kp; if (n_kp==0) zero++;
        esp_camera_fb_return(fb);
    }
    st->avg = (float)(sum/nframes);
    const double var = (sum2/nframes) - (st->avg*st->avg);
    st->std = (float)((var>0)?sqrt(var):0);
    st->zero = zero;
}

static void auto_sweep_choose(void)
{
#if AUTO_EXPO_SWEEP
    sensor_set_auto(false, false);

    sweep_stat_t best = {.avg=-1e9f,.std=1e9f,.zero=9999,.aec=0,.agc=0};
    sweep_stat_t cur;

    for (size_t i=0;i<sizeof((int[]){220,300,380,460})/sizeof(int); ++i){
        for (size_t j=0;j<sizeof((int[]){8,12,16,24})/sizeof(int); ++j){
            sensor_set_aec(((int[]){220,300,380,460})[i]);
            sensor_set_agc_gain(((int[]){8,12,16,24})[j]);
            vTaskDelay(pdMS_TO_TICKS(60));
            cur.aec=((int[]){220,300,380,460})[i];
            cur.agc=((int[]){8,12,16,24})[j];
            run_frames_eval(12, &cur);

            ESP_LOGI(TAG, "SWEEP aec=%d agc=%d  avg=%.1f std=%.1f zero=%d",
                     cur.aec, cur.agc, cur.avg, cur.std, cur.zero);

            bool better = (cur.avg > best.avg + 0.5f) ||
                          ((fabsf(cur.avg - best.avg) < 0.5f) && (cur.zero < best.zero)) ||
                          ((fabsf(cur.avg - best.avg) < 0.5f) && (cur.zero == best.zero) && (cur.std < best.std));
            if (better) best = cur;
        }
    }
    sensor_set_aec(best.aec);
    sensor_set_agc_gain(best.agc);
    ESP_LOGI(TAG, "LOCK aec=%d agc=%d  (avg corners=%.1f, std=%.1f)",
             best.aec, best.agc, best.avg, best.std);

#if LOCK_AE_AFTER_SWEEP
    sensor_set_auto(false, false);  // lock both
#else
    sensor_set_auto(true, true);
#endif
#endif
}

typedef struct {
    uint16_t x, y;       // integer pixel coords
    uint16_t score0;     // FAST score at seed
    int8_t   vx, vy;     // last accepted displacement (motion prior)
    uint8_t  age;        // frames
    uint8_t  alive;      // 0/1
#if USE_BRIEF
    uint32_t desc[4];    // 128-bit BRIEF
#else
    uint32_t desc[3];    // 96-bit Census
#endif
} Track;

#define MAX_TRACKS         160
#define SOFT_CAP_ALIVE      80
#define TARGET_TRACKS      140

static int   g_ntracks = 0;

// per-frame search radius (set in main loop)
static int   g_win_rad = WIN_RAD_DEFAULT;

// ----------------------- Descriptor Builders --------------------------------
static inline int popc32(uint32_t x){ return __builtin_popcount(x); }

#if !USE_BRIEF
static void build_census9x9(const uint8_t* img, int w, int h, int x, int y, uint32_t out[3])
{
    const int cx = x, cy = y;
    const uint8_t ref = img[cy*w + cx];
    uint32_t b0=0,b1=0,b2=0;
    int bit=0;
    for (int dy=-4; dy<=4; ++dy){
        for (int dx=-4; dx<=4; ++dx){
            if (dx==0 && dy==0) continue;
            uint8_t v = img[(cy+dy)*w + (cx+dx)];
            uint32_t bitval = (v > ref);
            if      (bit < 32) b0 |= (bitval << bit);
            else if (bit < 64) b1 |= (bitval << (bit-32));
            else               b2 |= (bitval << (bit-64));
            bit++;
        }
    }
    out[0]=b0; out[1]=b1; out[2]=b2; // (last 16 bits padded)
}
static inline int ham96(const uint32_t a[3], const uint32_t b[3]){
    return popc32(a[0]^b[0]) + popc32(a[1]^b[1]) + popc32(a[2]^b[2]);
}
#else
static const int8_t BRIEF_PAIRS[128][4] = {{0,0,0,0}}; // TODO
static void build_brief128(const uint8_t* img, int w, int h, int x, int y, uint32_t out[4])
{
    uint32_t b0=0,b1=0,b2=0,b3=0;
    for (int i=0;i<128;i++){
        int x1 = x + BRIEF_PAIRS[i][0], y1 = y + BRIEF_PAIRS[i][1];
        int x2 = x + BRIEF_PAIRS[i][2], y2 = y + BRIEF_PAIRS[i][3];
        uint8_t p1 = img[y1*w + x1], p2 = img[y2*w + x2];
        uint32_t bit = (p1>p2);
        if      (i<32)  b0 |= (bit<<i);
        else if (i<64)  b1 |= (bit<<(i-32));
        else if (i<96)  b2 |= (bit<<(i-64));
        else            b3 |= (bit<<(i-96));
    }
    out[0]=b0; out[1]=b1; out[2]=b2; out[3]=b3;
}
static inline int ham128(const uint32_t a[4], const uint32_t b[4]){
    return popc32(a[0]^b[0]) + popc32(a[1]^b[1]) + popc32(a[2]^b[2]) + popc32(a[3]^b[3]);
}
#endif

// -------------------------- Utils -------------------------------------------
static inline bool in_bounds_seed(int x,int y){
    return (x>=SEED_BORDER && x<g_w-SEED_BORDER && y>=SEED_BORDER && y<g_h-SEED_BORDER);
}
static inline int l2_sq(int x1,int y1,int x2,int y2){
    int dx=x1-x2, dy=y1-y2; return dx*dx+dy*dy;
}
// tiny frame texture score (stride-8 abs gradient)
static inline int frame_texture_score(const uint8_t* img, int w, int h){
    int sum = 0, cnt = 0;
    for (int y=4; y<h-4; y+=8){
        const int yw = y*w;
        for (int x=4; x<w-4; x+=8){
            int gx = img[yw + x+1] - img[yw + x-1];
            int gy = img[(y+1)*w + x] - img[(y-1)*w + x];
            int g  = (gx<0?-gx:gx) + (gy<0?-gy:gy);
            sum += g; cnt++;
        }
    }
    return (cnt>0)? (sum/cnt) : 0;
}
static inline int clampi(int v,int lo,int hi){ return v<lo?lo:(v>hi?hi:v); }
static inline float clampf(float v,float lo,float hi){ return v<lo?lo:(v>hi?hi:v); }

// -------------------------- Tracker core ------------------------------------
static void tracker_reset(void){
    memset(g_tracks, 0, sizeof(g_tracks));
    g_ntracks = 0;
}

// avoid seeding too close to existing tracks (r=3 px)
static bool too_close_to_tracks(int x,int y){
    for (int i=0;i<g_ntracks;i++){
        if (!g_tracks[i].alive) continue;
        if (l2_sq(x,y, g_tracks[i].x, g_tracks[i].y) <= 9) return true;
    }
    return false;
}

// Seed new tracks from ranked FAST list (returns added)
static int tracker_seed_from_fast(const Keypoint* kps, int n, const uint8_t* img,
                                  int max_add, int seed_min_score)
{
    int added=0;
    for (int i=0;i<n && added<max_add; ++i){
        const Keypoint *k = &kps[i];
        if (k->score < seed_min_score) continue;
        if (!in_bounds_seed(k->x, k->y)) continue;
        if (too_close_to_tracks(k->x, k->y)) continue;

        Track t = {0};
        t.x = k->x; t.y = k->y; t.score0 = k->score; t.age = 0; t.alive = 1; t.vx=0; t.vy=0;
    #if USE_BRIEF
        build_brief128(img, g_w, g_h, t.x, t.y, t.desc);
    #else
        build_census9x9(img, g_w, g_h, t.x, t.y, t.desc);
    #endif

        int slot = -1;
        for (int j=0;j<g_ntracks;j++) if (!g_tracks[j].alive){ slot=j; break; }
        if (slot>=0) g_tracks[slot]=t;
        else {
            if (g_ntracks < MAX_TRACKS) g_tracks[g_ntracks++] = t;
            else break;
        }
        added++;
    }
    return added;
}

// compact to keep alive densely packed
static void tracker_compact(void){
    int w=0;
    for (int i=0;i<g_ntracks;i++){
        if (g_tracks[i].alive) {
            if (i!=w) g_tracks[w]=g_tracks[i];
            w++;
        }
    }
    g_ntracks = w;
}

// --- Match + FB --------------------------------------------------------------
static bool match_window_hamming(const uint8_t* prev, const uint8_t* curr, const Track* trk,
                                 int *mx, int *my, int *best_cost, int *second_cost, int ham_gate)
{
    const int cx = trk->x + trk->vx;
    const int cy = trk->y + trk->vy;
    int best=9999, second=9999, bx=trk->x, by=trk->y;

    for (int dy=-g_win_rad; dy<=g_win_rad; ++dy){
        int y = cy + dy; if (y<SEED_BORDER || y>=g_h-SEED_BORDER) continue;
        for (int dx=-g_win_rad; dx<=g_win_rad; ++dx){
            int x = cx + dx; if (x<SEED_BORDER || x>=g_w-SEED_BORDER) continue;
        #if USE_BRIEF
            uint32_t dcur[4]; build_brief128(curr, g_w, g_h, x, y, dcur);
            int cost = ham128(trk->desc, dcur);
        #else
            uint32_t dcur[3]; build_census9x9(curr, g_w, g_h, x, y, dcur);
            int cost = ham96(trk->desc, dcur);
        #endif
            if (cost < best){ second = best; best = cost; bx = x; by = y; }
            else if (cost < second){ second = cost; }
        }
    }

    *mx = bx; *my = by; *best_cost = best; *second_cost = second;

    if (best > ham_gate) return false;
    bool ratio_ok  = (second <= 0) || (best*100 <= 85*second);
    bool margin_ok = ((second - best) >= HAM_MARGIN_MIN);
    return (ratio_ok || margin_ok);
}

static bool fb_check(const uint8_t* prev, const uint8_t* curr, int x_prev, int y_prev, int x_curr, int y_curr, int ham_gate)
{
#if USE_BRIEF
    uint32_t dcur[4]; build_brief128(curr, g_w, g_h, x_curr, y_curr, dcur);
#else
    uint32_t dcur[3]; build_census9x9(curr, g_w, g_h, x_curr, y_curr, dcur);
#endif
    int best=9999, bx=x_curr, by=y_curr;

    for (int dy=-g_win_rad; dy<=g_win_rad; ++dy){
        int y = y_curr + dy; if (y<SEED_BORDER || y>=g_h-SEED_BORDER) continue;
        for (int dx=-g_win_rad; dx<=g_win_rad; ++dx){
            int x = y_curr + dx; (void)x; // quiet analyzer (x recomputed below)
            int xx = x_curr + dx;
            if (xx<SEED_BORDER || xx>=g_w-SEED_BORDER) continue;
        #if USE_BRIEF
            uint32_t dprev[4]; build_brief128(prev, g_w, g_h, xx, y, dprev);
            int cost = ham128(dcur, dprev);
        #else
            uint32_t dprev[3]; build_census9x9(prev, g_w, g_h, xx, y, dprev);
            int cost = ham96(dcur, dprev);
        #endif
            if (cost < best){ best = cost; bx = xx; by = y; }
        }
    }

    int dx = bx - x_prev, dy = by - y_prev;
    if (best > ham_gate) return false;
    return (dx*dx + dy*dy) <= (FB_THR_PX*FB_THR_PX);
}

// --------- Match buffer for Sim2 (accepted pairs + costs) -------------------
#define MATCH_BUF 256
static int16_t g_px[MATCH_BUF], g_py[MATCH_BUF], g_qx[MATCH_BUF], g_qy[MATCH_BUF];
static uint8_t g_cost[MATCH_BUF];
static int g_mcount = 0;

// Update all tracks; returns alive count and fills debug stats
typedef struct {
    int alive;
    int matched;
    int lost_ham;
    int lost_fb;
    int reseeded;
    int ham_sum, ham_min, ham_max;
    int best2_sum; // second-best accumulator
    // per-frame robust median stats for logging
    float med_dx, med_dy, med_abs_dx, med_abs_dy;
} TrkStats;

// helpers for medians
static int cmp_int(const void* a, const void* b){
    const int ai = *(const int*)a, bi = *(const int*)b;
    return (ai>bi) - (ai<bi);
}
static float median_of_ints(const int *a, int n){
    if (n<=0) return 0.f;
    int *tmp = (int*)alloca(n*sizeof(int));
    for (int i=0;i<n;i++) tmp[i] = a[i];
    qsort(tmp, n, sizeof(int), cmp_int);
    return (n & 1) ? (float)tmp[n/2] : 0.5f*(tmp[n/2 - 1] + tmp[n/2]);
}
static float median_of_abs(const int *a, int n){
    if (n<=0) return 0.f;
    int *tmp = (int*)alloca(n*sizeof(int));
    for (int i=0;i<n;i++){ int v=a[i]; tmp[i]= v<0?-v:v; }
    qsort(tmp, n, sizeof(int), cmp_int);
    return (n & 1) ? (float)tmp[n/2] : 0.5f*(tmp[n/2 - 1] + tmp[n/2]);
}

static int tracker_update(const uint8_t* prev, const uint8_t* curr, TrkStats* ts, int ham_gate)
{
    memset(ts, 0, sizeof(*ts));
    ts->ham_min = 9999;

    int dxs[MAX_TRACKS], dys[MAX_TRACKS]; int nd=0;

    for (int i=0;i<g_ntracks;i++){
        Track *t = &g_tracks[i];
        if (!t->alive) continue;

        int mx,my,best,second;
        bool ok = match_window_hamming(prev, curr, t, &mx,&my,&best,&second, ham_gate);
        if (!ok){ t->alive = 0; ts->lost_ham++; continue; }

        bool fb_ok = true;
        if (((t->age+1) % FB_EVERY) == 0){
            fb_ok = fb_check(prev, curr, t->x, t->y, mx, my, ham_gate);
        }
        if (!fb_ok){ t->alive=0; ts->lost_fb++; continue; }

        // Accept update
        int dx = mx - t->x;
        int dy = my - t->y;
        t->x = mx; t->y = my; t->age++;

        // update motion prior (clamped to [-3..3])
        if (dx < -3) dx = -3; else if (dx > 3) dx = 3;
        if (dy < -3) dy = -3; else if (dy > 3) dy = 3;
        t->vx = (int8_t)dx;
        t->vy = (int8_t)dy;

        // refresh descriptor at new position
        #if USE_BRIEF
          build_brief128(curr, g_w, g_h, t->x, t->y, t->desc);
        #else
          build_census9x9(curr, g_w, g_h, t->x, t->y, t->desc);
        #endif

        ts->matched++;
        ts->ham_sum += best;
        ts->best2_sum += (second>0?second:0);
        if (best < ts->ham_min) ts->ham_min = best;
        if (best > ts->ham_max) ts->ham_max = best;

        if (nd < MAX_TRACKS){ dxs[nd]=dx; dys[nd]=dy; nd++; }

        // --- stash pair for Sim2 -------------
        if (g_mcount < MATCH_BUF){
            g_px[g_mcount]   = (int16_t)(t->x - dx);
            g_py[g_mcount]   = (int16_t)(t->y - dy);
            g_qx[g_mcount]   = (int16_t)(t->x);
            g_qy[g_mcount]   = (int16_t)(t->y);
            g_cost[g_mcount] = (uint8_t)clampi(best,0,255);
            g_mcount++;
        }
    }

    // Compact + soft cap on alive tracks
    tracker_compact();
    int alive=0;
    for (int i=0;i<g_ntracks;i++){
        if (g_tracks[i].alive){
            if (alive >= SOFT_CAP_ALIVE) { g_tracks[i].alive = 0; }
            else alive++;
        }
    }
    ts->alive = alive;
    tracker_compact();

    // robust medians for logging
    ts->med_dx     = (nd>0)? median_of_ints(dxs, nd)    : 0.f;
    ts->med_dy     = (nd>0)? median_of_ints(dys, nd)    : 0.f;
    ts->med_abs_dx = (nd>0)? median_of_abs(dxs, nd)     : 0.f;
    ts->med_abs_dy = (nd>0)? median_of_abs(dys, nd)     : 0.f;

    return ts->alive;
}

// ========================= Sim2 from point pairs (RANSAC) ===================
typedef struct { float s, theta, tx, ty; float inlier_frac; float rmse; } Sim2Fit;

static inline void sim2_from_2pts(float x1,float y1,float x2,float y2,
                                  float X1,float Y1,float X2,float Y2,
                                  float* s,float* th,float* tx,float* ty)
{
    float dxp = x2-x1, dyp = y2-y1;
    float dxq = X2-X1, dyq = Y2-Y1;
    float rp = sqrtf(dxp*dxp + dyp*dyp) + 1e-6f;
    float rq = sqrtf(dxq*dxq + dyq*dyq) + 1e-6f;
    float scale = rq/rp;
    float angp = atan2f(dyp, dxp);
    float angq = atan2f(dyq, dxq);
    float theta = angq - angp;
    float c = cosf(theta), si = sinf(theta);
    float tx0 = X1 - scale*(c*x1 - si*y1);
    float ty0 = Y1 - scale*(si*x1 + c*y1);
    *s = scale; *th = theta; *tx = tx0; *ty = ty0;
}

static Sim2Fit sim2_ransac_PQ(const int16_t *px,const int16_t *py,
                               const int16_t *qx,const int16_t *qy,
                               const uint8_t *cost, int N, float eps_px,
                               Sim2Fit prev)
{
    Sim2Fit best={.s=1.f,.theta=0.f,.tx=0.f,.ty=0.f,.inlier_frac=-1.f,.rmse=0.f};
    if (N<4) return best;

    // optional trim by cost (keep best 75%)
#if RANSAC_TRIM_Q25
    static uint8_t tmpc[MATCH_BUF];
    for (int i=0;i<N;i++) tmpc[i] = cost[i];
    // sort copy to find cutoff
    for (int i=0;i<N-1;i++) for (int j=i+1;j<N;j++) if (tmpc[j]<tmpc[i]) { uint8_t t=tmpc[i]; tmpc[i]=tmpc[j]; tmpc[j]=t; }
    uint8_t cutoff = tmpc[(3*N)/4];
#endif

    const int iters = 64;
    for (int it=0; it<iters; ++it){
        // sample two distinct indices
        int i1 = esp_random() % N, i2 = esp_random() % N; if (i2==i1) i2 = (i1+1)%N;

#if RANSAC_TRIM_Q25
        if (cost[i1]>cutoff || cost[i2]>cutoff) continue;
#endif

        float s,th,tx,ty;
        sim2_from_2pts(px[i1],py[i1], px[i2],py[i2], qx[i1],qy[i1], qx[i2],qy[i2], &s,&th,&tx,&ty);

        float c=cosf(th), si=sinf(th);
        int inl=0; double se2=0.0;
        for (int k=0;k<N;k++){
#if RANSAC_TRIM_Q25
            if (cost[k]>cutoff) continue;
#endif
            float xp = s*(c*px[k] - si*py[k]) + tx;
            float yp = s*(si*px[k] + c*py[k]) + ty;
            float ex = xp - qx[k];
            float ey = yp - qy[k];
            float e2 = ex*ex + ey*ey;
            if (e2 <= eps_px*eps_px){ inl++; se2 += e2; }
        }
        float frac = (float)inl / (float)N;
        if (frac > best.inlier_frac){
            best.s=s; best.theta=th; best.tx=tx; best.ty=ty;
            best.inlier_frac = frac;
            best.rmse = (inl>0)? sqrtf((float)(se2/(double)inl)) : 0.f;
        }
    }

    // rotation clamp with hysteresis
    if (fabsf(best.theta)*(180.0f/(float)M_PI) > ROT_CLAMP_DEG &&
        !(best.inlier_frac > prev.inlier_frac + ROT_IMPROVE_MIN))
    {
        best.theta = prev.theta; // keep previous rotation
    }
    return best;
}

// ------------------------------- APP MAIN -----------------------------------
void app_main(void)
{
    ESP_ERROR_CHECK(camera_init_gray());
    auto_sweep_choose();

    // Warm prev buffer
    for (int i=0;i<2;i++){
        camera_fb_t *fb = esp_camera_fb_get();
        if (fb){ memcpy(g_prev, fb->buf, g_w*g_h); esp_camera_fb_return(fb); }
    }
    tracker_reset();

    uint32_t frames=0; int64_t t0=esp_timer_get_time();

    // motion accumulators for log
    float cum_dx=0.f, cum_dy=0.f;
    // reseed hygiene cooldown
    int rh_cool=0;
    // previous Sim2 for hysteresis
    Sim2Fit prev_fit = {.s=1.f,.theta=0.f,.tx=0.f,.ty=0.f,.inlier_frac=-1.f,.rmse=0.f};

    while (true) {
        camera_fb_t *fb = esp_camera_fb_get();
        if (!fb) { vTaskDelay(pdMS_TO_TICKS(1)); continue; }

        // Low-texture score for this frame
        int tex = frame_texture_score(fb->buf, g_w, g_h);
        bool lowtex = (tex < 12);  // gentler threshold

        // Per-frame search radius
        g_win_rad = lowtex ? WIN_RAD_LOW_TEX : WIN_RAD_DEFAULT;

        // Fixed absolute ham gate (+1 when lowtex)
        int ham_gate = HAM_ABS_MAX + (lowtex ? 1 : 0);

        // Reseed governance (every RESEED_EVERY frames) — never skip; just throttle
        int n_kp = -1, reseeded = 0;
        if ((frames % RESEED_EVERY) == 0){
            int alive_now=0; for (int i=0;i<g_ntracks;i++) if (g_tracks[i].alive) alive_now++;

            if (alive_now < TARGET_TRACKS){
                fast_detect_and_nms(fb->buf, g_w, g_h, &n_kp);
                g_last_fast_cnt = n_kp;

                int cap  = lowtex ? RESEED_CAP_LOW : RESEED_CAP_PERF;

#if RESEED_HYGIENE_ON
                if (alive_now > RH_ALIVE_MIN && tex < RH_TEX_MAX){
                    cap = (cap>2)?2:cap;
                    rh_cool = RH_FRAMES;            // keep cap for a couple frames
                }
                if (rh_cool>0){ cap = (cap>2)?2:cap; rh_cool--; }
#endif
                int want = TARGET_TRACKS - alive_now;
                if (want > cap) want = cap;

                if (want > 0 && n_kp > 0){
                    int seed_min = lowtex ? SEED_MIN_LOW : SEED_MIN_SCORE;
                    reseeded = tracker_seed_from_fast(keypoints, n_kp, fb->buf, want, seed_min);
                }
            }
        }

        // Track prev -> curr
        TrkStats ts;
        int alive_now = tracker_update(g_prev, fb->buf, &ts, ham_gate);
        ts.reseeded = reseeded;

        // update cumulative from robust medians
        cum_dx += ts.med_dx;
        cum_dy += ts.med_dy;

        // Once-per-second stats (+ Sim2 RANSAC on P,Q pairs)
        frames++;
        int64_t now = esp_timer_get_time();
        float dt_s = (now - t0)*1e-6f;

        if (dt_s >= 1.0f) {
            float fps = frames/dt_s;
            int ham_avg = (ts.matched>0) ? (ts.ham_sum/ts.matched) : -1;
            int best2_avg = (ts.matched>0 && ts.best2_sum>0) ? (ts.best2_sum/ts.matched) : (ts.matched?0:-1);

            // Sim2 from gathered matches
            Sim2Fit fit = {.s=1.f,.theta=0.f,.tx=0.f,.ty=0.f,.inlier_frac=-1.f,.rmse=0.f};
            if (g_mcount >= 4){
                // eps based on current motion estimate
                float m = fabsf(ts.med_abs_dx) > fabsf(ts.med_abs_dy) ? ts.med_abs_dx : ts.med_abs_dy;
                float eps = clampf(0.6f + 0.3f*m, 0.8f, 2.0f); // 0.8..2.0 px
                fit = sim2_ransac_PQ(g_px,g_py,g_qx,g_qy,g_cost,g_mcount, eps, prev_fit);
                prev_fit = fit;
            }
            float th_deg = fit.theta * (180.0f/(float)M_PI);

            char fast_buf[12]; const char *fast_field = "--";
            if (g_last_fast_cnt >= 0){ snprintf(fast_buf,sizeof(fast_buf),"%d", g_last_fast_cnt); fast_field = fast_buf; }

#if LOG_INCHES
            float cum_dx_in = -cum_dx / PX_PER_IN;
            float cum_dy_in = -cum_dy / PX_PER_IN;
            ESP_LOGI(TAG,
                "FPS=%.1f FAST=%s Tracks: alive=%d, matched=%d, reseeded=%d, "
                "rej_ham=%d, rej_fb=%d, ham[min/avg/max]=[%d/%d/%d], 2nd_best_avg=%d, tex=%d, sr=%d | "
                "Model=%s inlier=%.2f s=%.3f th=%.2fdeg t=(%.2f,%.2f) rmse=%.3f | "
                "med(dx,dy)=%.1f,%.1f medAbs=%.1f,%.1f cum=(%.1f,%.1f)px ≈ (%.2f,%.2f)in",
                fps, fast_field, alive_now, ts.matched, ts.reseeded,
                ts.lost_ham, ts.lost_fb,
                (ts.matched?ts.ham_min:-1), ham_avg, (ts.matched?ts.ham_max:-1),
                best2_avg, tex, g_win_rad,
                (fit.inlier_frac<0?"none":"Sim2"), fit.inlier_frac<0?-1.0f:fit.inlier_frac,
                fit.s, th_deg, fit.tx, fit.ty, fit.rmse,
                ts.med_dx, ts.med_dy, ts.med_abs_dx, ts.med_abs_dy,
                cum_dx, cum_dy, cum_dx_in, cum_dy_in);
#else
            ESP_LOGI(TAG,
                "FPS=%.1f FAST=%s Tracks: alive=%d, matched=%d, reseeded=%d, "
                "rej_ham=%d, rej_fb=%d, ham[min/avg/max]=[%d/%d/%d], 2nd_best_avg=%d, tex=%d, sr=%d | "
                "Model=%s inlier=%.2f s=%.3f th=%.2fdeg t=(%.2f,%.2f) rmse=%.3f | "
                "med(dx,dy)=%.1f,%.1f medAbs=%.1f,%.1f cum=(%.1f,%.1f)",
                fps, fast_field, alive_now, ts.matched, ts.reseeded,
                ts.lost_ham, ts.lost_fb,
                (ts.matched?ts.ham_min:-1), ham_avg, (ts.matched?ts.ham_max:-1),
                best2_avg, tex, g_win_rad,
                (fit.inlier_frac<0?"none":"Sim2"), fit.inlier_frac<0?-1.0f:fit.inlier_frac,
                fit.s, th_deg, fit.tx, fit.ty, fit.rmse,
                ts.med_dx, ts.med_dy, ts.med_abs_dx, ts.med_abs_dy,
                cum_dx, cum_dy);
#endif

            // reset per-second buffers
            frames=0; t0=now; g_last_fast_cnt = -1;
            g_mcount = 0;
        }

        // Roll frame (curr → prev)
        memcpy(g_prev, fb->buf, g_w*g_h);
        esp_camera_fb_return(fb);
        vTaskDelay(pdMS_TO_TICKS(1));
    }
}
